# COMMANDS FOR DEVELOPMENTS


1. gcc main.c src/*.c -Iinclude -o obj/gpacalc --> [command used to compile "main" executable]

2. gcc test/test_scrconv.c src/scrconv.c -Iinclude -o obj/test_scrconv --> [command used to compile "test_scrconv" executable]

3. ./obj/<executable file name> --> [Git bash command to run compile program]

4. \obj\<executable file name> --> [CMD / Powershell command to run compile program]

5. gcc test/test_getch.c src/getch.c -Iinclude -o obj/test_getch --> [command used to compile "test_getch" executable]

6. ./obj/test_getch.exe < test/test_getch_input.txt --> [Git bash command to run compile test program]

7. gcc test/test_getcrs.c src/getcrs.c -Iinclude -o obj/test_getcrs --> [command used to compile "test_getcrs" executable]

8. ./obj/test_getcrs.exe < test/test_getcrs_input.txt --> [Git bash command to run compile test program]
